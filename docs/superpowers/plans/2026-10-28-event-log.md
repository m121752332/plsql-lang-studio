# 事件紀錄與事件檢視器 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 用 Loguru＋SQLite 記錄操作過程與錯誤，提供依等級設定保留天數的「配置」視窗與可篩選、匯出的事件檢視器。

**Architecture:** 新模組 `pllangstudio/eventlog.py`（不依賴 Qt）掛一個 Loguru sink，把每筆紀錄寫進 `%APPDATA%\PLLangStudio\events.db`，並提供查詢、清除、匯出、未預期例外攔截。`MainWindow` 在各操作的成功與失敗處呼叫它；錯誤對話框附上事件編號。新的 `SettingsDialog`（外觀、紀錄事件兩頁）取代編輯選單的主題與字型，`EventViewer` 是獨立視窗。

**Tech Stack:** Python 3.14、PySide6、Loguru、sqlite3（標準庫）、pytest＋pytest-qt＋xdist、uv。

**Spec:** [docs/superpowers/specs/2026-10-28-event-log-design.md](../specs/2026-10-28-event-log-design.md)

## Global Constraints

- 一律用 uv：`uv add loguru`、`uv run pytest`；不可直接呼叫系統 `python`／`pip`。
- `core/` 不可 import PySide6；`eventlog.py` 也不可 import PySide6。
- 介面文字、註解、ADR 用繁體中文（台灣用語），遵守 CONTEXT.md 用詞（主語系／副語系、語言檔、錯誤、警告……）。
- 含中文的檔案只能用 Write／Edit 工具修改，不可用 `sed -i`、heredoc、`echo` 寫入。
- 測試必須能平行執行：資料庫一律放 `tmp_path`，測試結束呼叫 `eventlog.shutdown()`；不可寫入真實的 `%APPDATA%` 或 Languages 目錄。
- 保留天數預設：資訊 90、警告 90、錯誤 30；範圍 1～3650；app.json 欄位 `log_days_info`、`log_days_warning`、`log_days_error`。
- 時間在資料庫存 **UTC** ISO 8601（毫秒），顯示時轉本地時間（比 spec 的「本地時間」更利於比較，Task 8 會同步更新 spec）。
- 紀錄本身失敗不可讓主程式出錯。
- Commit：AGENTS.md 規定未經使用者要求不 commit。每個 Task 結尾的 commit 步驟**只在使用者同意後執行**；訊息用 Conventional Commits＋繁體中文，結尾加 `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`。

```python
"""事件紀錄：用 Loguru 記錄操作過程與錯誤，存進 SQLite，供事件檢視器查詢與匯出（見 ADR-0017）。

紀錄本身失敗時（資料庫壞掉、被鎖、沒有權限）一律吞掉，不可影響主程式。
"""

from __future__ import annotations

import os
import sqlite3
import threading
import traceback
from dataclasses import dataclass
from datetime import datetime, timezone
from pathlib import Path

from loguru import logger

from . import __version__, appconfig

GROUP_OF_LEVEL = {"TRACE": "info", "DEBUG": "info", "INFO": "info", "SUCCESS": "info",
                  "WARNING": "warning", "ERROR": "error", "CRITICAL": "error"}
GROUP_LABEL = {"info": "資訊", "warning": "警告", "error": "錯誤"}
LEVEL_LABEL = {"TRACE": "資訊", "DEBUG": "資訊", "INFO": "資訊", "SUCCESS": "資訊",
               "WARNING": "警告", "ERROR": "錯誤", "CRITICAL": "嚴重錯誤"}

_SCHEMA = """
CREATE TABLE IF NOT EXISTS events (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    time TEXT NOT NULL,
    level TEXT NOT NULL,
    category TEXT NOT NULL DEFAULT '',
    message TEXT NOT NULL,
    detail TEXT NOT NULL DEFAULT '',
    app_version TEXT NOT NULL DEFAULT ''
);
CREATE INDEX IF NOT EXISTS events_time ON events(time);
CREATE INDEX IF NOT EXISTS events_level ON events(level);
"""
_COLUMNS = "id, time, level, category, message, detail, app_version"

_lock = threading.Lock()
_conn: sqlite3.Connection | None = None
_db_path: Path | None = None
_handler_id: int | None = None
_last_id: int | None = None


@dataclass(frozen=True)
class Event:
    id: int
    time: datetime  # 本地時間
    level: str
    category: str
    message: str
    detail: str
    app_version: str

    @property
    def level_label(self) -> str:
        return LEVEL_LABEL.get(self.level, self.level)


def _utc(dt: datetime) -> str:
    """存成 UTC 字串，字串大小順序就是時間順序；沒有時區的視為本地時間。"""
    return dt.astimezone(timezone.utc).isoformat(timespec="milliseconds")


def _open(path: Path) -> sqlite3.Connection:
    path.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(path, check_same_thread=False, isolation_level=None)  # 自動提交
    try:
        conn.execute("PRAGMA journal_mode=WAL")  # 檢視器讀取時不擋住寫入
        conn.executescript(_SCHEMA)
        if conn.execute("PRAGMA quick_check").fetchone()[0] != "ok":
            raise sqlite3.DatabaseError("quick_check 失敗")
    except sqlite3.Error:
        conn.close()
        raise
    return conn


def setup(db_path: Path, retention: dict[str, int] | None = None) -> None:
    """啟用事件紀錄。資料庫壞了就改名成 .corrupt 再重建；開不了就只是不記錄。"""
    global _conn, _db_path, _handler_id
    shutdown()
    db_path = Path(db_path)
    try:
        try:
            conn = _open(db_path)
        except sqlite3.DatabaseError:
            appconfig.quarantine(db_path)
            for suffix in ("-wal", "-shm"):
                db_path.with_name(db_path.name + suffix).unlink(missing_ok=True)
            conn = _open(db_path)
    except (OSError, sqlite3.Error):
        conn = None
    _conn, _db_path = conn, db_path
    logger.remove()  # 視窗程式沒有主控台，不要 Loguru 預設的 stderr 輸出
    _handler_id = logger.add(_sink, level="DEBUG", catch=False)


def shutdown() -> None:
    """停止記錄並關閉資料庫（測試結束、程式結束時呼叫）。"""
    global _conn, _db_path, _handler_id
    if _handler_id is not None:
        logger.remove(_handler_id)
        _handler_id = None
    with _lock:
        if _conn is not None:
            _conn.close()
        _conn, _db_path = None, None


def _sink(message) -> None:
    global _last_id
    r = message.record
    detail = ""
    if r["exception"] is not None:
        t, v, tb = r["exception"]
        detail = "".join(traceback.format_exception(t, v, tb))
    row = (_utc(r["time"]), r["level"].name, str(r["extra"].get("category", "")), r["message"], detail, __version__)
    with _lock:
        _last_id = None
        if _conn is None:
            return
        try:
            _last_id = _conn.execute(
                "INSERT INTO events (time, level, category, message, detail, app_version) VALUES (?, ?, ?, ?, ?, ?)",
                row).lastrowid
        except sqlite3.Error:
            pass


def log(level: str, category: str, message: str, exc=None) -> int | None:
    """寫一筆事件，回傳事件編號；沒有啟用或寫入失敗時回傳 None。exc 可以是例外物件或 (type, value, tb)。"""
    global _last_id
    _last_id = None
    logger.bind(category=category).opt(exception=exc, depth=1).log(level, message)
    return _last_id


def info(category: str, message: str) -> int | None:
    return log("INFO", category, message)


def success(category: str, message: str) -> int | None:
    return log("SUCCESS", category, message)


def warning(category: str, message: str) -> int | None:
    return log("WARNING", category, message)


def error(category: str, message: str, exc=None) -> int | None:
    return log("ERROR", category, message, exc)


def critical(category: str, message: str, exc=None) -> int | None:
    return log("CRITICAL", category, message, exc)


def _like(text: str) -> str:
    return "%" + text.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_") + "%"


def _event(row) -> Event:
    eid, time, level, category, message, detail, version = row
    return Event(eid, datetime.fromisoformat(time).astimezone(), level, category, message, detail, version)


def query(groups=("info", "warning", "error"), since: datetime | None = None, keyword: str = "",
          limit: int = 5000) -> list[Event]:
    """查詢事件，新的在前。keyword 是 #數字 時只依事件編號找，忽略其他條件。"""
    kw = keyword.strip()
    if kw.startswith("#") and kw[1:].isdigit():
        sql, args = f"SELECT {_COLUMNS} FROM events WHERE id = ?", [int(kw[1:])]
    else:
        levels = [lv for lv, g in GROUP_OF_LEVEL.items() if g in groups]
        if not levels:
            return []
        where, args = [f"level IN ({','.join('?' * len(levels))})"], list(levels)
        if since is not None:
            where.append("time >= ?")
            args.append(_utc(since))
        if kw:
            where.append("(message LIKE ? ESCAPE '\\' OR category LIKE ? ESCAPE '\\' OR detail LIKE ? ESCAPE '\\')")
            args += [_like(kw)] * 3
        sql = f"SELECT {_COLUMNS} FROM events WHERE {' AND '.join(where)} ORDER BY id DESC LIMIT ?"
        args.append(limit)
    with _lock:
        if _conn is None:
            return []
        try:
            rows = _conn.execute(sql, args).fetchall()
        except sqlite3.Error:
            return []
    return [_event(r) for r in rows]
```

```python
    """事件紀錄各等級群組的保留天數；沒設定或不合法時用預設值。"""
    data = _load()
    result = {}
    for group, default in LOG_RETENTION_DEFAULT.items():
```

```python
def purge(retention: dict[str, int], now: datetime | None = None) -> int:
    """依各等級群組的保留天數刪除過期紀錄，回傳刪除筆數。"""
    now = now or datetime.now(timezone.utc)
    removed = 0
    with _lock:
        if _conn is None:
            return 0
        try:
            for group, days in retention.items():
                levels = [lv for lv, g in GROUP_OF_LEVEL.items() if g == group]
                removed += _conn.execute(
                    f"DELETE FROM events WHERE time < ? AND level IN ({','.join('?' * len(levels))})",
                    [_utc(now - timedelta(days=days)), *levels]).rowcount
        except sqlite3.Error:
            pass
    return removed


def stats() -> tuple[int, int]:
    """回傳（總筆數, 資料庫檔案大小 bytes，含 -wal）。"""
    with _lock:
        if _conn is None or _db_path is None:
            return 0, 0
        try:
            count = _conn.execute("SELECT COUNT(*) FROM events").fetchone()[0]
        except sqlite3.Error:
            count = 0
        size = sum(p.stat().st_size for p in (_db_path, _db_path.with_name(_db_path.name + "-wal")) if p.exists())
    return count, size
```

```python
def format_event(e: Event) -> str:
    head = f"#{e.id}  {e.time:%Y-%m-%d %H:%M:%S}  {e.level_label}  [{e.category}]  {e.message}"
    return head + (f"\n{e.detail.rstrip()}" if e.detail else "")


def system_info(extra: dict[str, str] | None = None) -> dict[str, str]:
    """回報問題時附上的環境資訊；Qt 版本由畫面層透過 extra 帶入（本模組不 import Qt）。"""
    info = {"程式版本": __version__, "Python": platform.python_version(), "作業系統": platform.platform()}
    info.update(extra or {})
    return info


def export_name(now: datetime) -> str:
    return f"PLLangStudio-events-{now:%Y%m%d-%H%M%S}.zip"


def export_zip(events: list[Event], dest: Path, extra: dict[str, str] | None = None) -> None:
    """匯出成 zip（events.txt＋system.txt），不含語言檔內容。先寫 .tmp 再換檔。"""
    lines = [f"{k}：{v}" for k, v in system_info(extra).items()]
    lines += [f"匯出時間：{datetime.now():%Y-%m-%d %H:%M:%S}", f"筆數：{len(events)}"]
    buf = io.BytesIO()
    with zipfile.ZipFile(buf, "w", zipfile.ZIP_DEFLATED) as z:
        z.writestr("events.txt", "\n\n".join(format_event(e) for e in events) + "\n")
        z.writestr("system.txt", "\n".join(lines) + "\n")
    langfile.atomic_write(Path(dest), buf.getvalue())


def with_event_id(text: str, event_id: int | None) -> str:
    """錯誤對話框的訊息附上事件編號，回報問題時用。"""
    return text if event_id is None else f"{text}\n\n事件編號 #{event_id}"


def crash_message(event_id: int | None) -> str:
    text = "發生未預期的錯誤。"
    if event_id is not None:
        text += f"（事件編號 #{event_id}）\n\n回報問題時請提供此編號，並從「配置 ▸ 事件檢視器」匯出紀錄。"
    return text


def install_excepthook(on_crash: Callable[[int | None], None]) -> None:
    """未預期的例外記成嚴重錯誤，再交給 on_crash 顯示事件編號。"""
    def hook(exc_type, exc, tb):
        if issubclass(exc_type, KeyboardInterrupt):
            sys.__excepthook__(exc_type, exc, tb)
            return
        eid = critical("未預期的錯誤", f"{exc_type.__name__}: {exc}", (exc_type, exc, tb))
        try:
            on_crash(eid)
        except Exception:
            pass
    sys.excepthook = hook
```

```python
"""事件檢視器：依等級、期間、關鍵字查詢事件紀錄，看詳細內容、複製與匯出（見 ADR-0017）。"""

from typing import Callable

from PySide6.QtGui import QColor
from PySide6.QtWidgets import (QAbstractItemView, QApplication, QCheckBox, QComboBox, QFileDialog, QHBoxLayout,

                               QTableWidget, QTableWidgetItem, QVBoxLayout, QWidget)

from .. import eventlog
from .theme import ThemeManager, colors

PERIODS = [("今天", "today"), ("最近 7 天", 7), ("最近 30 天", 30), ("全部", None)]
HEADERS = ["編號", "時間", "等級", "類別", "訊息"]


class EventViewer(QWidget):
    def __init__(self, theme: ThemeManager | None = None, parent=None,
                 save_path: Callable[[str], str] | None = None):
        super().__init__(parent)
        self.setWindowFlag(Qt.Window)  # 獨立視窗，可以邊操作主視窗邊看
        self.setWindowTitle("事件檢視器")
        self.resize(1000, 600)
        self.save_path = save_path or self._ask_save_path
        self.events: list[eventlog.Event] = []

        top = QHBoxLayout()
        top.addWidget(QLabel("等級"))
        self.level_boxes: dict[str, QCheckBox] = {}
        for group, label in eventlog.GROUP_LABEL.items():
            box = QCheckBox(label)
            box.setChecked(True)
            box.toggled.connect(self.refresh)
            top.addWidget(box)
            self.level_boxes[group] = box
        top.addSpacing(20)
        top.addWidget(QLabel("期間"))
        self.period = QComboBox()
        for label, value in PERIODS:
            self.period.addItem(label, value)
        self.period.setCurrentIndex(1)
        self.period.currentIndexChanged.connect(self.refresh)
        top.addWidget(self.period)
        self.search = QLineEdit()
        self.search.setPlaceholderText("搜尋訊息或類別；輸入 #編號 直接找事件")
        self.search.textChanged.connect(self.refresh)
        top.addWidget(self.search, 1)
        refresh_button = QPushButton("重新整理")
        refresh_button.clicked.connect(self.refresh)
        top.addWidget(refresh_button)

        self.table = QTableWidget(0, len(HEADERS))
        self.table.setHorizontalHeaderLabels(HEADERS)
        self.table.setSelectionBehavior(QAbstractItemView.SelectRows)
        self.table.setEditTriggers(QAbstractItemView.NoEditTriggers)
        self.table.verticalHeader().hide()
        self.table.horizontalHeader().setStretchLastSection(True)
        for col, w in enumerate((70, 150, 70, 100)):
            self.table.setColumnWidth(col, w)
        self.table.itemSelectionChanged.connect(self._show_detail)
        self.detail = QPlainTextEdit()
        self.detail.setReadOnly(True)
        split = QSplitter(Qt.Vertical)
        split.addWidget(self.table)
        split.addWidget(self.detail)
        split.setStretchFactor(0, 3)

        bottom = QHBoxLayout()
        self.count_label = QLabel()
        bottom.addWidget(self.count_label, 1)
        copy_button = QPushButton("複製選取")
        copy_button.clicked.connect(self.copy_selected)
        export_button = QPushButton("匯出篩選結果…")
        export_button.clicked.connect(self.export)
        bottom.addWidget(copy_button)
        bottom.addWidget(export_button)

        lay = QVBoxLayout(self)
        lay.addLayout(top)
        lay.addWidget(split, 1)
        lay.addLayout(bottom)
        if theme is not None:
            theme.changed.connect(self.refresh)
        self.refresh()

    def since(self) -> datetime | None:
        value = self.period.currentData()
        now = datetime.now()
        if value == "today":
            return now.replace(hour=0, minute=0, second=0, microsecond=0)
        return None if value is None else now - timedelta(days=value)

    def refresh(self, *_args) -> None:
        groups = {g for g, b in self.level_boxes.items() if b.isChecked()}
        self.events = eventlog.query(groups, self.since(), self.search.text())
        level_fg = colors().event_level
        self.table.setRowCount(len(self.events))
        for r, e in enumerate(self.events):
            color = level_fg.get(eventlog.GROUP_OF_LEVEL.get(e.level, ""))
            first_line = e.message.splitlines()[0] if e.message else ""
            for c, text in enumerate((str(e.id), f"{e.time:%Y-%m-%d %H:%M:%S}", e.level_label, e.category, first_line)):
                item = QTableWidgetItem(text)
                if color:
                    item.setForeground(QColor(color))
                self.table.setItem(r, c, item)
        self.count_label.setText(f"共 {len(self.events):,} 筆")
        if self.events:
            self.table.selectRow(0)
        else:
            self.detail.clear()

    def selected_events(self) -> list[eventlog.Event]:
        rows = sorted({i.row() for i in self.table.selectedIndexes()})
        return [self.events[r] for r in rows if r < len(self.events)]

    def _show_detail(self) -> None:
        selected = self.selected_events()
        self.detail.setPlainText(eventlog.format_event(selected[0]) if selected else "")

    def copy_selected(self) -> None:
        QApplication.clipboard().setText("\n\n".join(eventlog.format_event(e) for e in self.selected_events()))

    def _ask_save_path(self, name: str) -> str:
        path, _ = QFileDialog.getSaveFileName(self, "匯出事件紀錄", str(Path.home() / "Desktop" / name), "zip 檔 (*.zip)")
        return path

    def export(self) -> None:
        if not self.events:
            return
        path = self.save_path(eventlog.export_name(datetime.now()))
        if not path:
            return
        try:
            eventlog.export_zip(self.events, Path(path), {"Qt": qVersion(), "PySide6": PySide6.__version__})
        except OSError as e:
            QMessageBox.warning(self, "匯出失敗", f"{path}\n\n{e}")
            return
        eventlog.info("事件檢視器", f"匯出 {len(self.events)} 筆事件紀錄到 {path}")
        self.count_label.setText(f"已匯出 {len(self.events):,} 筆到 {path}")
```

```python
from PySide6.QtWidgets import QApplication, QMessageBox

from pllangstudio.core.workspace import default_root

def main() -> int:
    if sys.platform == "win32":  # 用 python 執行時，工作列才會顯示本程式的圖示而不是 Python 的

    app.aboutToQuit.connect(lambda: eventlog.info("程式", "PLLangStudio 結束"))
    theme = ThemeManager(app)  # 先套用主題和字型再建立視窗，避免開啟時閃一下

    root = appconfig.last_workspace() or default_root(appconfig.app_dir())
    win = MainWindow(root, theme_manager=theme, font_manager=fonts)
    win.show()
    code = app.exec()
    eventlog.shutdown()
    return code


if __name__ == "__main__":
    sys.exit(main())
```

```markdown
# 用 Loguru＋SQLite 記錄操作與錯誤，事件編號用來回報問題

程式突然出錯時，原本只有對話框，關掉就沒了，無法追查使用者之前做了什麼。我們加入**事件紀錄**：記錄每個操作的結果與所有錯誤，存在 `%APPDATA%\PLLangStudio\events.db`（SQLite）。每筆有**事件編號**，錯誤對話框會顯示它；回報問題時提供編號，並從**事件檢視器**匯出 zip。

做法：

- `pllangstudio/eventlog.py`（不依賴 Qt）掛一個 Loguru sink，同步寫入 SQLite（WAL 模式），寫入後即可取得事件編號。
- 記錄操作層級：開啟／切換工作區、匯入、移除、切換主語系／副語系、存檔、部署、批次修改、用語對照表、記事本重新載入、配置變更、程式啟動與結束。**不記錄字串的翻譯內容**，匯出的 zip 也不含語言檔。
- 等級分三群保留：資訊（DEBUG／INFO／SUCCESS）90 天、警告 90 天、錯誤（ERROR／CRITICAL）30 天，可在「配置 ▸ 紀錄事件」調整（1～3650 天），存在 app.json。啟動時與套用配置後清除過期紀錄。
- 時間存 UTC，顯示時轉本地時間。
- 未預期的例外由 `sys.excepthook` 攔截，記為嚴重錯誤並顯示事件編號。
- 紀錄本身失敗（資料庫損毀、被鎖、沒有權限）不可影響主程式；損毀的資料庫改名為 `.corrupt` 後重建（比照 app.json）。

## Considered Options

- **標準 `logging`**：不需要額外套件，但 Loguru 的 API 較簡潔、例外堆疊較完整。採用 Loguru。
- **只寫文字檔**：查詢、依等級清除、取得事件編號都不方便。不採用。
- **記錄每個字串的修改內容**：追查最完整，但紀錄量大，且翻譯內容會出現在使用者回報的紀錄裡。不採用；需要時可從 backup 資料夾比對。
```
